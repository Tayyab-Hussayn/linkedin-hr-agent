// Find the top-level message wrappers (parent of user/claude message divs)
const userMsgs = document.querySelectorAll('div[class*="font-user-message"]');
const claudeMsgs = document.querySelectorAll('div[class*="font-claude-response"]');

console.log("User messages found:", userMsgs.length);
console.log("Claude messages found:", claudeMsgs.length);

// Get parent containers (one level up = full turn block)
const allTurns = new Set();
userMsgs.forEach(el => { if (el.parentElement) allTurns.add(el.parentElement); });
claudeMsgs.forEach(el => { if (el.parentElement) allTurns.add(el.parentElement); });

const turnsArray = Array.from(allTurns);
console.log("Total turns detected:", turnsArray.length);

// Remove all but last 6
const toRemove = turnsArray.slice(0, -6);
toRemove.forEach(el => el.remove());

console.log(`Removed ${toRemove.length} turns`);
console.log("Remaining nodes:", document.querySelectorAll('*').length);
