(function cleanClaude() {
  // 1. Disable animations
  const s = document.createElement('style');
  s.textContent = `* { animation: none !important; transition: none !important; }`;
  document.head.appendChild(s);

  // 2. Strip syntax highlighting spans
  document.querySelectorAll('pre code').forEach(pre => {
    pre.textContent = pre.textContent;
  });

  // 3. Kill hidden/aria elements & SVGs
  document.querySelectorAll('[aria-hidden="true"], svg').forEach(el => el.remove());

  // 4. Nuke old messages from main container (keep last 20)
  const conv = document.querySelector('div.flex-1.flex.flex-col.px-4.max-w-3xl.mx-auto.w-full.pt-1');
  if (conv) {
    const children = Array.from(conv.children);
    children.slice(0, -20).forEach(el => el.remove());
  }

  console.log("✅ Done. Remaining nodes:", document.querySelectorAll('*').length);
})();
